    {
        public AdminPage(User selectedUser)
        {
            InitializeComponent();
            DataContext = selectedUser;
            UserGrid.ItemsSource = DB.Context.User.Where(u => u.id != 1).ToList();
        }
        private void BtnEdit_Click(object sender, RoutedEventArgs e)
        {
            if (((sender as Button).DataContext as User).admin == true)
            {
                ((sender as Button).DataContext as User).admin = false;
                ((sender as Button).DataContext as User).salesman = true;
            }
            else if (((sender as Button).DataContext as User).salesman == true)
            {
                ((sender as Button).DataContext as User).salesman = false;
                ((sender as Button).DataContext as User).admin = false;
            }
            else
            {
                ((sender as Button).DataContext as User).admin = true;
                ((sender as Button).DataContext as User).salesman = false;
            }
            DB.Context.SaveChanges();
            UserGrid.ItemsSource = null;
            UserGrid.ItemsSource = DB.Context.User.Where(u => u.id != 1).ToList();
        }
        private void Exit_Click(object sender, RoutedEventArgs e)
        {
            NavigationService.Navigate(new RegisterPage());
        }
    }

    /// </summary>
    public partial class AutorisationPage : Page
    {
        private User _currentUser = new User();
        public AutorisationPage()
        {
            InitializeComponent();
            DataContext = _currentUser;
        }

        private void Auto_Click(object sender, RoutedEventArgs e)
        {
            StringBuilder errors = new StringBuilder();
            if (string.IsNullOrWhiteSpace(_currentUser.login))
                errors.AppendLine("Укажите Login");
            if (string.IsNullOrWhiteSpace(_currentUser.password))
                errors.AppendLine("Укажите Пароль");
            var existingUser = DB.Context.User.FirstOrDefault(u => u.login == _currentUser.login && u.password == _currentUser.password);
            if (existingUser == null)
                errors.AppendLine("Неверный Логин или Пароль");
            if (errors.Length > 0)
            {
                MessageBox.Show(errors.ToString(), "Ошибка");
                return;
            }
            try
            {
                if (existingUser.admin == false)
                    NavigationService.Navigate(new LKPage(selectedUser: existingUser));
                else if (existingUser.salesman == true)
                    NavigationService.Navigate(new SalesmanPage(selectedUser:existingUser));
                else
                    NavigationService.Navigate(new AdminPage(selectedUser: existingUser));
                MessageBox.Show("Добро пожаловать " + existingUser.fio, "Сообщение", MessageBoxButton.OK, MessageBoxImage.Information);
            }
            catch (Exception ex)
            {
                MessageBox.Show(ex.Message.ToString(), "Сообщение", MessageBoxButton.OK, MessageBoxImage.Information);
            }
        }

        private void Register_Click(object sender, RoutedEventArgs e)
        {
            NavigationService.Navigate(new RegisterPage());
        }
    }
}

    /// <summary>
    /// Логика взаимодействия для OrderPage.xaml
    /// </summary>
    public partial class OrderPage : Page
    {
        Order order = new Order();
        public OrderPage(User selectedUser)
        {
            InitializeComponent();
            DataContext = order;
            order.UserId = selectedUser.id;
            order.Status = "Оформлен";
            ComboProducts.ItemsSource = DB.Context.Product.ToList();
        }

        private void SaveBtn_Click(object sender, RoutedEventArgs e)
        {
            StringBuilder errors = new StringBuilder();
            if (order.Product.name == null)
                errors.AppendLine("Выберите продукт");
            if (order.Count <=0)
                errors.AppendLine("Укажите количество");
            if (errors.Length > 0)
            {
                MessageBox.Show(errors.ToString());
                return;
            }


            if (order.id == 0)
            {
                order.id = DB.Context.Order.Max(p => p.id) + 1;
                DB.Context.Order.Add(order);
            }
            try
            {
                order.Total = order.Product.price * order.Count;
                DB.Context.SaveChanges();
                MessageBox.Show("Заказ создан!");
                NavigationService.GoBack();
            }
            catch (Exception ex)
            {
                MessageBox.Show(ex.Message.ToString());
            } 
        }
    }
}
